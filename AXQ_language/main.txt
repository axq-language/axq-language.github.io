int axq__append_ptr(uintptr_t **max_ptr,long *main_ptr){
    if(max_ptr==NULL || *max_ptr==NULL){
        printf("\033[31mSerious error\n");
        printf("The variable is corrupted.\033[0m\n");
        interrupted = 1;
        return 1;
    }
    (*max_ptr)[0]=(uintptr_t)calloc(2,sizeof(uintptr_t));
    if((*max_ptr)[0]==0){
        printf("\033[31mSerious error\n");
        printf("Dynamic memory allocation failed.\033[0m\n");
        interrupted = 1;
        return 1;
    }
    *max_ptr=(uintptr_t *)(*max_ptr)[0];
    (*max_ptr)[1]=(uintptr_t)main_ptr;
    return 0;
}

long* axq__find_ptr(long *ptr,long *inout,int inpt_s){
    if(inpt_s==0){
        return ptr;
    }
    if(inpt_s<=0){
        printf("\033[31merror\nInvalid dimension count (0).\033[0m\n");
        interrupted = 1;
        return NULL;
    }
    long *buf;
    long *ptr_buf=ptr;
    long count=0;
    long max;
    if(ptr == NULL || inout == NULL){
        printf("\033[31mSerious error\n");
        printf("axq Internal Error: Root pointer is NULL.\033[0m\n");
        interrupted = 1;
        return NULL;
    }
    for(int i=0;i<inpt_s-1;i++){
        max=ptr_buf[5];
        if(max<inout[i]){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            return NULL;
        }
        for(count=0;count<(inout[i]-(inout[i]%1000000));count+=1000000){
            ptr_buf=(long*)(uintptr_t)ptr_buf[7];
        }
        for(count=count;count<(inout[i]-(inout[i]%10000));count+=10000){
            ptr_buf=(long*)(uintptr_t)ptr_buf[6];
        }
        for(count=count;count<(inout[i]-(inout[i]%100));count+=100){
            ptr_buf=(long*)(uintptr_t)ptr_buf[4];
        }
        while(count<inout[i]){
            ptr_buf=(long*)(uintptr_t)ptr_buf[0];
            count++;
        }
        //移動完了
        //イメージ：特急で駅を飛ばしながら近いところまで行き、そこから各駅停車でジリジリ進み、駅まで。
        //11/7 ルイボいない　別の日に
        //12/20 プレゼン資料など諸々、締め切り
        if(count!=inout[i]){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
        if(ptr_buf==NULL){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
        if(ptr_buf[1]!=0){
            ptr_buf=(long *)(uintptr_t)ptr_buf[1];
        }else{
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
    }
    for(count=0;count<(inout[inpt_s-1]-(inout[inpt_s-1]%1000000));count+=1000000){
        ptr_buf=(long*)(uintptr_t)ptr_buf[7];
        if(ptr_buf==NULL){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
    }
    for(count=count;count<(inout[inpt_s-1]-(inout[inpt_s-1]%10000));count+=10000){
        ptr_buf=(long*)(uintptr_t)ptr_buf[6];
        if(ptr_buf==NULL){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
    }
    for(count=count;count<(inout[inpt_s-1]-(inout[inpt_s-1]%100));count+=100){
        ptr_buf=(long*)(uintptr_t)ptr_buf[4];
        if(ptr_buf==NULL){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
    }
    while(count<inout[inpt_s-1]){
        ptr_buf=(long*)(uintptr_t)ptr_buf[0];
        if(ptr_buf==NULL){
            printf("\033[31merror\n");
            for(int j=0;j<inpt_s;j++){
                printf("[%ld]",inout[j]);
            }
            printf("not found.\033[0m\n");
            interrupted = 1;
            return NULL;
        }
        count++;
    }
    if(ptr_buf[0]==0){
        axq__next_zero=1;
    }else{
        axq__next_zero=0;
    }
    return ptr_buf;
}

int axq__append(long *ptr,uintptr_t **ptrs,long *inout,long inpt_s,double data_do,char data_ch,int mode){
    if(mode<1 || mode>3){
        return 1;
    }
    //append(a[10])
    long* point_ptr;
    long* buffer;
    long max=0;
    long count=0;
    int flag_I=0;
    int flag_II=0;
    int flag_III=0;
    long* I_ptr;
    long* II_ptr;
    long* III_ptr;
    long* datas;
    if(ptr==NULL){
        return 1;
    }
    if(ptr[5]!=0){
        point_ptr=axq__find_ptr(ptr,inout,inpt_s);
        if(point_ptr==NULL){
            return 1;
        }
        if(inpt_s!=0){
            buffer=(long *)(uintptr_t)point_ptr[1];
            if(buffer==NULL){
                return 1;
            }
            point_ptr=buffer;//目標が存在する階層に移動した。
        }
        if(point_ptr[5]==0){
            point_ptr[5]=1;
            point_ptr[3]=mode;
            if(mode==1){
                memcpy(&point_ptr[2], &data_do, sizeof data_do);
            }else if(mode==2){
                memcpy(&point_ptr[2], &data_ch, sizeof data_ch);
            }else if(mode==3){
                point_ptr[2]=0;
            }
            return 0;
        }else{
            max=point_ptr[5]-1;
            datas=point_ptr;
            for(count=0;count<max-(max%1000000);count+=1000000){
                point_ptr=(long*)(uintptr_t)point_ptr[7];
            }
            if((max+1)%1000000==0){
                I_ptr=point_ptr;
                flag_I=1;
            }
            for(count=count;count<max-(max%10000);count+=10000){
                point_ptr=(long*)(uintptr_t)point_ptr[6];
            }
            if((max+1)%10000==0){
                II_ptr=point_ptr;
                flag_II=1;
            }
            for(count=count;count<(max-(max%100));count+=100){
            point_ptr=(long*)(uintptr_t)point_ptr[4];
        }
    }
    if((max+1)%100==0){
        III_ptr=point_ptr;
        flag_III=1;
    }
    //分割
    //次のポインタを記憶する。
    //    0       100     199 200
    //    100     200     ......

    //    0       10000
    //    10000   20000   ......

    //    0       1000000
    //    1000000 2000000 ......
    while(count<max){
        point_ptr=(long*)(uintptr_t)point_ptr[0];
        count++;
    }
    if(mode==1){//double
        if(point_ptr==NULL){
            return 1;
        }
        //次のポインタを作る
        point_ptr[0]=(long)(uintptr_t)calloc(8,sizeof(long));
        if(point_ptr[0]==0){
            return 1;
        }
        if(axq__append_ptr(ptrs,(long*)point_ptr[0])==1){
            free ((uintptr_t*)point_ptr[0]);
            point_ptr[0]=0;
            return 1;
        }
        if(flag_III==1){
            III_ptr[4]=point_ptr[0];
        }
        if(flag_II==1){
            II_ptr[6]=point_ptr[0];
        }
        if(flag_I==1){
            I_ptr[7]=point_ptr[0];
        }
        memcpy(&((long *)(uintptr_t)point_ptr[0])[2], &data_do, sizeof data_do);
        ((long *)(uintptr_t)point_ptr[0])[3]=1;
        datas[5]++;
        }else if(mode==2){
            if(point_ptr==NULL){
                return 1;
            }
            //次のポインタを作る
            point_ptr[0]=(long)(uintptr_t)calloc(8,sizeof(long));
            if(point_ptr[0]==0){
                return 1;
            }
            if(axq__append_ptr(ptrs,(long*)point_ptr[0])==1){
                free ((uintptr_t*)point_ptr[0]);
                point_ptr[0]=0;
                return 1;
            }
            if(flag_III==1){
                III_ptr[4]=point_ptr[0];
            }
            if(flag_II==1){
                II_ptr[6]=point_ptr[0];
            }
            if(flag_I==1){
                I_ptr[7]=point_ptr[0];
            }
            memcpy(&((long *)(uintptr_t)point_ptr[0])[2], &data_ch, sizeof data_ch);
            ((long *)(uintptr_t)point_ptr[0])[3]=2;
            datas[5]++;
        }else if(mode==3){
            if(point_ptr==NULL){
                return 1;
            }
            //次のポインタを作る
            point_ptr[0]=(long)(uintptr_t)calloc(8,sizeof(long));
            if(point_ptr[0]==0){
                return 1;
            }
            if(axq__append_ptr(ptrs,(long*)point_ptr[0])==1){
                free ((uintptr_t*)point_ptr[0]);
                point_ptr[0]=0;
                return 1;
            }
            if(flag_III==1){
                III_ptr[4]=point_ptr[0];
            }
            if(flag_II==1){
                II_ptr[6]=point_ptr[0];
            }
            if(flag_I==1){
                I_ptr[7]=point_ptr[0];
            }
            datas[5]++;
            ((long *)(uintptr_t)point_ptr[0])[3]=3;
        }
    }else{
        ptr[5]=1;
        ptr[3]=mode;
        if(mode==1){
            memcpy(&ptr[2], &data_do, sizeof data_do);
        }else if(mode==2){
            memcpy(&ptr[2], &data_ch, sizeof data_ch);
        }else if(mode==3){
            ptr[2]=0;
        }
    }
    return 0;
}

int axq__free(uintptr_t *ptr,long *ptr_free,long num){
    uintptr_t *ptrptr=ptr;
    uintptr_t *buf;
    uintptr_t *buffer;
    if(ptr==NULL){
        printf("\033[31mSerious error\naxq pointer not found.\033[0m\n");
        return 1;
    }
    if(ptr_free[num]){
        return 0;
    }
    while(ptr[0]!=0){
        buffer=(uintptr_t*)ptr[0];
        buf=(uintptr_t*)ptr[1];
        free(buf);
        free_count++;
        ptr=buffer;
    }
    buffer=(uintptr_t*)ptr[0];
    buf=(uintptr_t*)ptr[1];
    free(buf);
    free_count++;
    while(ptrptr[0]!=0){
        buffer=(uintptr_t*)ptrptr[0];
        free(ptrptr);
        ptrptr=buffer;
    }
    buffer=(uintptr_t*)ptrptr[0];
    free(ptrptr);
    ptr_free[num]=1;
    return 0;
}

int axq__print(long *ptr, long *inout, long inpt_s) {
    if(ptr == NULL || inout == NULL || inpt_s <= 0) {
        return 1;
    }
    long *node = axq__find_ptr(ptr, inout, inpt_s);
    if(node == NULL) {
        return 1;
    }
    long path[inpt_s + 1];
    for(long j = 0; j < inpt_s; j++) {
        path[j] = inout[j];
    }
    int first = 1;
    int in_string = 0;
    printf("[");
    for(long i = 0;; i++) {
        path[inpt_s - 1] = i;
        node = axq__find_ptr(ptr, path, inpt_s);
        if(node == NULL) {
            return 1;
        }
        /* 再帰呼び出しで変わる前に末尾情報を保存 */
        int is_last = (node[0] == 0);
        int is_array = (node[1] != 0);
        int type = (int)node[3];
        if(is_array || type == 1 || type == 2) {
            if(type == 2 && !is_array) {
                if(!in_string) {
                    if(!first) {
                        printf(", ");
                    }
                    printf("\'");
                    in_string = 1;
                    first = 0;
                }
                char ch = axq__char_return(node[2]);
                if(ch != '\0') {
                    putchar(ch);
                }
            }else{
                if(in_string) {
                    printf("\'");
                    in_string = 0;
                }
                if(!first) {
                    printf(", ");
                }
                if(is_array) {
                    path[inpt_s] = 0;

                    if(axq__print(ptr, path, inpt_s + 1)) {
                        return 1;
                    }
                }else{
                    printf("%lf", axq__double_return(node[2]));
                }
                first = 0;
            }
        }
        if(is_last) {
            break;
        }
    }
    if(in_string) {
        printf("\'");
    }
    printf("]");
    return 0;
}
int axq__string_change(long* ptr,uintptr_t **ptrs,long* inout,long inpt_s,char *data,long data_s){
    long i;
    long *buf = axq__find_ptr(ptr,inout,inpt_s);
    if(buf==NULL){
        return 1;
    }
    long aaaaa=buf[5];
    axq__next_zero=0;
    long max=buf[5];
    for(i=0;i<data_s && i<aaaaa;i++){
        inout[inpt_s-1]=i;
        if(axq__vi_set(ptr,inout,inpt_s,data[i])){
            return 1;
        }
        axq__type_change(ptr,inout,inpt_s,2);
    }
    for(i=i;i<data_s;i++){
        inout[inpt_s-1]=i;
        if(axq__append(ptr,ptrs,inout,inpt_s-1,0,data[i],2)==1){
            return 1;
        }
    }
    for(i=i;i<max;i++){
        inout[inpt_s-1]=i;
        if(axq__vi_set(ptr,inout,inpt_s,'\0')){
            return 1;
        }
    }
    return 0;
}
//ptrのデータをptr_cにコピー
int axq__cpy(
long *ptr, long *inout, long inpt_s, uintptr_t **max,
long *ptr_c, long *inout_c, long inpt_s_c, uintptr_t **max_c,
long count_) {
    if(inpt_s<1){
        printf("error\n");
        return 1;
    }
    if(inpt_s_c<1){
        printf("error\n");
        return 1;
    }
    if(ptr == NULL || inout == NULL || inpt_s <= 0) {
        return 1;
    }
    long *node = axq__find_ptr(ptr, inout, inpt_s);
    if(node == NULL) {
        return 1;
    }
    long path[inpt_s + 1];
    for(long j = 0; j < inpt_s; j++) {
        path[j] = inout[j];
    }
    long path_c[inpt_s_c + 1];
    for(long j = 0; j < inpt_s_c; j++) {
        path_c[j] = inout_c[j];
    }
    path_c[inpt_s_c]=0;
    int first = 1;
    int in_string = 0;
    if(0<count_){
        if(axq__vi_ap(ptr_c,inout_c,inpt_s_c-1,max_c)){
            return 1;
        }
    }
    for(long i = 0;; i++) {
        path[inpt_s - 1] = i;
        path_c[inpt_s_c - 1] = i;
        node = axq__find_ptr(ptr, path, inpt_s);
        if(node == NULL) {
            return 1;
        }
        /* 再帰呼び出しで変わる前に末尾情報を保存 */
        int is_last = (node[0] == 0);
        int is_array = (node[1] != 0);
        int type = (int)node[3];
        if(is_array || type == 1 || type == 2) {
            if(type == 2 && !is_array) {
                if(axq__append(ptr_c,max_c,inout_c,inpt_s_c-1,0,axq__char_return(node[2]),2)){
                    return 1;
                }
            }else{
                if(is_array) {
                    path[inpt_s] = 0;
                    if(axq__cpy(ptr, path, inpt_s + 1,max,ptr_c,path_c,inpt_s_c+1,max_c,count_+1)) {
                        return 1;
                    }
                }else{
                    if(axq__append(ptr_c,max_c,inout_c,inpt_s_c-1,axq__double_return(node[2]),'\0',1)){
                        return 1;
                    }
                }
                first = 0;
            }
        }
        if(is_last) {
            break;
        }
    }
    return 0;
}
